# 最终复现命令清单

## 1. 主实验（正式）
cd ~/AFLplusplus
export NV_TOKEN='你的真实token'
ENDPOINTS="person_detail review_count hotpic_list cms_doc_list" DUR=120 ./scripts/run_main_baseline_rule_generic.sh
cat out/main_baseline_rule_generic/summary.csv

## 2. 二级 score 服务启动
cd ~/AFLplusplus
pkill -f nv_valid_server_mock.py || true
rm -f /tmp/nv_valid.sock
export NV_RPC_FMT=text
export NV_VALID_SOCK=/tmp/nv_valid.sock
python3 nv_valid_server_mock.py

## 3. 二级 score 手工验证
cd ~/AFLplusplus
export NV_TARGET_CONFIG=$PWD/targets/o2oa_query.json
export NV_TOKEN='你的真实token'
export NV_ENDPOINT_NAME='cms_doc_list'
export NV_BODY_RULES=$PWD/validity/o2oa_query_rules.json
export NV_BODY_SCORE_ENDPOINT='unix:///tmp/nv_valid.sock'
export NV_BODY_SCORE_THRESHOLD='1.5'
export NV_BODY_VALID_STATS=/tmp/nv_body_valid_stats.json
export NV_DEBUG_BODY_VALID=1

for f in \
  in/o2oa_body_cms_score/seed_ok_0.json \
  in/o2oa_body_cms_score/seed_border_4.json \
  in/o2oa_body_cms_score/seed_badscore_3.json \
  in/o2oa_body_cms_score/seed_badscore_7.json
do
  echo "==== $f ===="
  python3 nv_http_harness.py < "$f"
done

## 4. 二级 score 增强实验
cd ~/AFLplusplus
export NV_TOKEN='你的真实token'

IN_DIR=$HOME/AFLplusplus/in/o2oa_body_cms_score DUR=120 ./scripts/run_cms_body_valid_compare.sh
