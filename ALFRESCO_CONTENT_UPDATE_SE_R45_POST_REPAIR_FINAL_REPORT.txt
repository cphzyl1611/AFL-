            # This handles missing NV_TOKEN, ALFRESCO_USER, ALFRESCO_PASS, etc.

                    json.dump({
                        "exec_seq": 0,       # Invalid identity per AFL++ contract
                        "http_code": -99,    # Special code for auth config error

                        "ncov_delta": 0,
                        "ncov_total": 0,
                        "nall": 0,
                        "is_exception": 1,   # Mark as exception
                        "recover_ms": 0
                    }, f)
                sys.exit(2)  # Exit code 2 for auth error
            raise
